# Pergunta 2 — Como é a distribuição da idade dos pacientes? - A idade é simétrica ou assimétrica? - Existem outliers? - A população é jovem, idosa ou mista?

dados %>% 
  filter(!is.na(AGE) & AGE != "") %>% 
  ggplot(aes(x = AGE)) +
  geom_histogram(bins = 30, color = "black", fill = "lightblue") +
  theme_minimal() +
  labs(title = "Distribuição da Idade", x = "Idade (anos)", y = "Contagem")

dados %>% 
  filter(!is.na(AGE) & AGE != "") %>% 
  ggplot(aes(y = AGE)) +
  geom_boxplot(fill = "lightblue") +
  theme_minimal() +
  labs(title = "Boxplot da Idade", y = "Idade (anos)")

# Pergunta 3 — Qual a distribuição de estágios tumorais? - Quantos pacientes estão nos estágios I, II, III, IV? - A amostra tem mais casos iniciais ou avançados?

dados %>% 
  filter(!is.na(STAGE) & STAGE != "") %>% 
  ggplot(aes(x = STAGE)) +
    geom_bar(fill = "steelblue") +
    theme_minimal() +
    labs(title = "Distribuição do Estágio", x = "Estágio", y = "Contagem")

Pergunta 4 — Qual é o perfil de tabagismo da coorte?

  • Quantos são nunca fumantes, ex-fumantes ou fumantes ativos?
  • Como está a distribuição de pack-years?
dados %>%
  filter(!is.na(SMOKING_HISTORY), SMOKING_HISTORY != "") %>%
  ggplot(aes(x = SMOKING_HISTORY)) +
  geom_bar(fill = "steelblue") +
  theme_minimal() +
  labs(
    title = "Distribuição Historico de Tabagismo",
    x = "Tipo de Fumante",
    y = "Número de Pacientes"
  ) +
  theme(axis.text.x = element_text(angle = 45, hjust = 1))
